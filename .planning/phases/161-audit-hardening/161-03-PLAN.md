```php
// ============================================================
// THE 6-FIELD CANONICAL — FROZEN — DO NOT DEVIATE
// ============================================================
// In audit:verify, $row comes from SELECT * FROM learning_audit_events ORDER BY seq_num ASC

$canonicalFields = [
    'seq'          => (int)$row['seq_num'],           // cast to int — matches write-time $newSeq
    'event_key'    => $row['event_key'],              // read from column directly
    'user_ref'     => $row['user_ref'],               // READ FROM COLUMN — NEVER recompute from user_id
                                                      // (DSGVO-01: user_id may be NULL after erasure;
                                                      //  user_ref is immutable and was stored at write time)
    'course_id'    => json_decode($row['context_json'], true)['course_id'] ?? null,
                                                      // no course_id column — extract from context_json ONCE
                                                      // (same as write-time: $context['course_id'] ?? null)
    'created_at'   => (int)$row['created_at'],        // cast to int — matches write-time $ts
    'payload_hash' => hash('sha256', $row['context_json']),
                                                      // hash the RAW stored string — NEVER decode+reencode
                                                      // (re-serialization changes key order and escaping)
];
ksort($canonicalFields);
// After ksort, alphabetical key order: course_id, created_at, event_key, payload_hash, seq, user_ref
$canonical = json_encode($canonicalFields, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES | JSON_THROW_ON_ERROR);
$expected  = hash('sha256', $canonical . '|' . $prevHash);
// $prevHash for seq_num=1 is str_repeat('0', 64) (genesis seed from Version009300)
// Compare $expected === $row['chain_hash']
// ============================================================
```

```
if $input->getOption('fork-runbook') → $output->writeln('docs/audit-fork-runbook.md') → return SUCCESS

PHASE 1 — CHAIN WALK:
  prevHash = str_repeat('0', 64)  // genesis seed
  fromSeq = (int)$input->getOption('from-seq') ?: 1
  SELECT * FROM learning_audit_events WHERE seq_num IS NOT NULL AND seq_num >= :fromSeq ORDER BY seq_num ASC

  For each $row:
    Reconstruct canonical using the FROZEN 6-field formula above (copy verbatim):
      $canonicalFields = [
          'seq'          => (int)$row['seq_num'],
          'event_key'    => $row['event_key'],
          'user_ref'     => $row['user_ref'],                          // READ FROM COLUMN, NEVER recompute
          'course_id'    => json_decode($row['context_json'], true)['course_id'] ?? null,
          'created_at'   => (int)$row['created_at'],
          'payload_hash' => hash('sha256', $row['context_json']),      // RAW stored string, NO decode+reencode
      ];
      ksort($canonicalFields);
      $canonical = json_encode($canonicalFields, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES | JSON_THROW_ON_ERROR);
      $expected  = hash('sha256', $canonical . '|' . $prevHash);

    If $expected !== $row['chain_hash']:
      Record failure: "Chain tampered at seq_num={$row['seq_num']}: expected {$expected}, stored {$row['chain_hash']}"
      $chainOk = false

    prevHash = $row['chain_hash']   // advance to next event

PHASE 2 — CHECKPOINT SIG VERIFY:
  SELECT * FROM learning_audit_checkpoints ORDER BY id ASC

  For each $checkpoint:
    Resolve public key via the mapper (NOT raw SQL):
      try { $certKey = $this->certKeyMapper->findByKeyId($checkpoint['key_id']); }
      catch (DoesNotExistException) { record failure "No key found for key_id {$checkpoint['key_id']}"; continue; }
    $pubKeyRaw = sodium_base642bin($certKey->getPublicKeyB64u(), SODIUM_BASE64_VARIANT_URLSAFE_NO_PADDING); // base64url → 32 raw bytes
    $sigRaw    = hex2bin($checkpoint['signature']);   // signature column IS hex (161-01: bin2hex)
    $valid = sodium_crypto_sign_verify_detached($sigRaw, $checkpoint['signed_payload'], $pubKeyRaw);
    If !$valid: record failure "Checkpoint {$checkpoint['id']} (to_event_id={$checkpoint['to_event_id']}): signature invalid"

    // Head hash consistency: checkpoint['head_hash'] must equal chain_hash of to_event_id row
    SELECT chain_hash FROM learning_audit_events WHERE seq_num = :to_event_id
    If mismatch: record failure "Checkpoint {$checkpoint['id']}: head_hash mismatch"

PHASE 3 — ANCHOR CONSISTENCY (optional, only if anchor_url IS NOT NULL):
  For each checkpoint where anchor_url IS NOT NULL:
    GET $checkpoint['anchor_url'] with timeout=5s
    If HTTP 200: parse JSON response, compare .content (base64_decode → compare to signed_payload)
    If mismatch or unreachable: record warning "Checkpoint {$id}: anchor mismatch or unreachable"
    (Anchor failure is WARNING not FAILURE — network may be temporarily unavailable)

OUTPUT:
  If no failures: print "✓ Chain intact ({N} events, {M} checkpoints verified)"  → return SUCCESS
  If failures:
    Print each failure line
    Print "Fork resolution runbook: docs/audit-fork-runbook.md"
    → return FAILURE
```
