    - The checkpoint DB row is always persisted before doForgejoAnchor() is called
    - forgejo_token value is NEVER written to a log line (log warning without the token)

Add `doForgejoAnchor(int $checkpointId, int $toEventId, string $signedPayload): ?string` to
`AuditCheckpointService`. This method is called by `createCheckpoint()` after the checkpoint row
is successfully INSERTed (not before — checkpoint is the source of truth, Forgejo is redundancy).

```php
private function doForgejoAnchor(int $checkpointId, int $toEventId, string $signedPayload): ?string {
    $enabled = $this->config->getAppValue('learning', 'forgejo_anchor_enabled', 'false') === 'true';
    $token   = $this->config->getAppValue('learning', 'forgejo_token', '');
    if (!$enabled || $token === '') {
        return null; // OFF by default — no HTTP, no error
    }

    $owner = $this->config->getAppValue('learning', 'forgejo_owner', '');
    $repo  = $this->config->getAppValue('learning', 'forgejo_repo', '');
    $path  = "audit-checkpoints/checkpoint-{$toEventId}.json";
    $url   = rtrim($this->config->getAppValue('learning', 'forgejo_base_url', ''), '/');
    $apiUrl = "{$url}/api/v1/repos/{$owner}/{$repo}/contents/{$path}";

    $body = json_encode([
        'message' => "audit checkpoint {$toEventId}",
        'content' => base64_encode($signedPayload),
    ], JSON_UNESCAPED_SLASHES | JSON_THROW_ON_ERROR);

    try {
        $ctx = stream_context_create(['http' => [
            'method'        => 'POST',
            'header'        => "Authorization: token {$token}\r\nContent-Type: application/json\r\n",
            'content'       => $body,
            'timeout'       => 10,
            'ignore_errors' => true,
        ]]);
        $response = @file_get_contents($apiUrl, false, $ctx);
        // $http_response_header is set by file_get_contents
        $statusLine = $http_response_header[0] ?? 'HTTP/1.1 0';
        preg_match('/HTTP\/\S+\s+(\d+)/', $statusLine, $m);
        $status = (int)($m[1] ?? 0);

        if ($status === 201 && $response !== false) {
            $data = json_decode($response, true);
            $htmlUrl = $data['content']['html_url'] ?? null;
            if ($htmlUrl !== null) {
                // Update anchor_url on the checkpoint row
                $qb = $this->db->getQueryBuilder();
                $qb->update('learning_audit_checkpoints')
                   ->set('anchor_url', $qb->createNamedParameter($htmlUrl))
                   ->where($qb->expr()->eq('id', $qb->createNamedParameter($checkpointId, \OCP\DB\QueryBuilder\IQueryBuilder::PARAM_INT)))
                   ->executeStatement();
                $this->config->setAppValue('learning', 'last_anchor_status', 'ok');
                return $htmlUrl;
            }
        }
        // Non-201 or missing html_url — soft-fail
        $this->logger->warning("Forgejo anchor failed (HTTP {$status}) for checkpoint {$checkpointId}", ['app' => 'learning']);
    } catch (\Throwable $e) {
        $this->logger->warning("Forgejo anchor exception for checkpoint {$checkpointId}: " . $e->getMessage(), ['app' => 'learning']);
    }

    $this->config->setAppValue('learning', 'last_anchor_status', 'failed');
    $this->config->setAppValue('learning', 'last_anchor_attempted_at', (string)time());
    return null; // anchor_url stays NULL on checkpoint row — never roll back checkpoint
}
```
