Entity contracts the later plans (02-07) consume — implement EXACTLY these:

CertKey (table learning_cert_keys):
  - id: int (autoincrement)
  - keyId: string (base64url public-key fragment, ≤64; == JWT kid fragment)
  - publicKeyB64u: string (JWK "x", base64url of 32-byte Ed25519 public key)
  - secretKeyEnc: string (ICrypto ciphertext of base64(64-byte secret key) — NEVER plaintext)
  - status: string ('active' | 'retired' | 'revoked')
  - createdAt: int (unix)
  jsonSerialize(): MUST return ['id','key_id','public_key_b64u','status','created_at'] — NEVER secret_key_enc.

Certificate (table learning_certificates):
  - id: int (autoincrement)
  - verificationId: string (UUIDv4, 36)
  - userId: string (≤64)
  - courseId: int
  - keyId: string (references CertKey.keyId)
  - credentialJson: text (the signed compact JWT — public, shareable)
  - revoked: bool (default false)
  - issuedAt: int (unix)
  - expiresAt: ?int (null = no expiry)

CertKeyMapper extends QBMapper: findByKeyId(string): CertKey; findActive(): ?CertKey;
  findAllNonRevoked(): CertKey[] (status != 'revoked' — active + retired).
CertificateMapper extends QBMapper: findByVerificationId(string): Certificate;
  findByUserAndCourse(string $userId, int $courseId): ?Certificate (idempotency guard for issuance);
  findByUserId(string): Certificate[].
