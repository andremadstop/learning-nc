    ```javascript
      nat:       () => import('./NatTable.vue'),
    ```

    ```javascript
    import firewallScenarios from '../../data/firewall_scenarios.json'

      authflow:  authflowScenarios,
    }
    ```

    **Props:**
    ```javascript
    props: {
      type:             { type: String, required: true },

      scenarioOverride: { type: Object, default: null },
      nodeId:           { type: String, default: '' },
    }
    ```

    **Computed:**
    ```javascript
    activeComponent() {
      return SIMULATOR_MAP[this.type] || null
    },
    resolvedScenario() {
      if (this.scenarioOverride) return this.scenarioOverride
      const list = SCENARIOS[this.type] || []
      return list.find(s => s.id === this.scenarioId) || null
    },
    ```

    **Methods — onResult Normierung (aus RESEARCH.md Pattern 5):**
    ```javascript
    onResult(rawResult) {
      // DNS free-lookup hat kein correct — kein @complete emittieren
      if (rawResult.kind === 'lookup') return

      let correct = false
      if (typeof rawResult.correct === 'boolean') {
        correct = rawResult.correct
      } else if (typeof rawResult.passed === 'boolean') {
        correct = rawResult.passed
      }
      const score = correct ? 1.0 : 0.0
      this.$emit('complete', correct, score, rawResult)
    },
    ```
