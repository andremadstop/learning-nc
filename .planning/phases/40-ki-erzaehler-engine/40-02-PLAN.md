   ```html
         @click="submitFreetext"
       >
         {{ freetextLoading ? '...' : '>' }}
       </button>
     </div>
     <p v-if="freetextError" class="ab-freetext-error">{{ freetextError }}</p>
   ```

   ```html
   </span>
   ```

4. **Data and methods in Vue component:**
   Add to `data()`:
   ```js
   freetextInput: '',
   freetextLoading: false,
   freetextError: '',
   ```

   Add method `submitFreetext()`:
   ```js
   async submitFreetext() {
     if (!this.freetextInput.trim() || this.freetextLoading) return;
     this.freetextLoading = true;
     this.freetextError = '';
     try {
       const { data } = await axios.post(
         generateUrl(`/apps/learning/api/story/campaigns/${this.activeCampaignId}/freetext`),
         { text: this.freetextInput.trim() }
       );
       if (data.valid && data.next_scene) {
         // Show narrative response, then load next scene
         this.freetextInput = '';
         this.showFreetextNarrative(data.narrative);
         // After narrative display, reload current scene (which is now advanced)
         await this.loadCurrentScene();
       } else {
         // Show the narrative as feedback (action was invalid or no next scene)
         this.freetextError = data.narrative || this.t('learning', 'Aktion nicht moeglich');
       }
     } catch (err) {
       this.freetextError = err.response?.data?.error || this.t('learning', 'Fehler bei der Verarbeitung');
     } finally {
       this.freetextLoading = false;
     }
   },
   ```

   Add method `showFreetextNarrative(text)` that temporarily displays the Gemini narrative response in the scene narrative area (like a typing effect), then transitions to the new scene. Re-use the existing `narrativeTyping` pattern if present, otherwise add a simple fade-in display.
