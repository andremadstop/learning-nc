Add reactive cert config state (initialized from course prop or API response):
```javascript
const certEnabled = ref(props.course?.certEnabled ?? false)
const certPassPercent = ref(props.course?.certPassPercent ?? 80)
const certRequiredPoolIds = ref(props.course?.certRequiredPoolIds ?? [])
const certValidityDays = ref(props.course?.certValidityDays ?? 0)
const certSaving = ref(false)

async function saveCertConfig() {
  certSaving.value = true
  try {
    const result = await updateCertConfig(props.course.id, {
      certEnabled: certEnabled.value,
      certPassPercent: certPassPercent.value,
      certRequiredPoolIds: certRequiredPoolIds.value,
      certValidityDays: certValidityDays.value,
    })
    // Update local state from server response
    certEnabled.value = result.certEnabled
    certPassPercent.value = result.certPassPercent
    certRequiredPoolIds.value = result.certRequiredPoolIds
    certValidityDays.value = result.certValidityDays
  } catch (e) {
    console.error('saveCertConfig failed', e)
  } finally {
    certSaving.value = false
  }
}
```

const passStatus = ref(null)
const passStatusLoading = ref(false)

async function fetchPassStatus() {
  if (!props.courseId) return
  passStatusLoading.value = true
  try {
    passStatus.value = await getPassStatus(props.courseId)
  } catch (e) {
    console.error('fetchPassStatus failed', e)
    passStatus.value = null
  } finally {
    passStatusLoading.value = false
  }
}
