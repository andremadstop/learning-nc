computed: {
  // Whether to show the Zeugnis card — false while loading or no data
  zeugnisVisible() {
    return this.passStatus !== null && !this.passStatusLoading
  },
  // Whether cert is applicable for this course (cert_enabled=true)
  certApplicable() {
    return this.passStatus?.applicable ?? false
  },
  // Whether the student has passed
  hasPassed() {
    return this.passStatus?.passed ?? false
  },
  // Formatted pass date string; null if not passed
  passedAtFormatted() {
    if (!this.passStatus?.passedAt) return null
    return new Date(this.passStatus.passedAt * 1000).toLocaleDateString()
  },
},
