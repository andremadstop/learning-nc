    PASSWORD STRATEGY (security-critical):
    - The synchronous path (≤50 rows): generate a random password per user with bin2hex(random_bytes(12)) — prints plaintext password to the output terminal ONCE per user with "User {uid}: {pw}" then the string is gone from memory. The plaintext is passed to IUserManager::createUser($uid, $pw) — NC hashes it internally. Never store plaintext anywhere.
    - The async path (>50 rows): ImportUsersJob generates its OWN random password per user inside run(). The job arg 'csv_data' contains ONLY username/display_name/email columns — NO password column. The admin must distribute NC password-reset links to users after bulk import. Print a note to the console: "Dispatched bulk import job. Users will be created with random passwords. Use NC admin panel to send password-reset emails."
    - CRITICAL: Never serialize passwords into IJobList::add() argument array — job args persist in oc_jobs DB table.

    Create app/lib/Command/ImportUsersCommand.php:
    ```php
    <?php
    declare(strict_types=1);
    namespace OCA\Learning\Command;

    use OC\Core\Command\Base;
    use OCA\Learning\Service\AuditService;
    use OCP\BackgroundJob\IJobList;
    use OCP\IGroupManager;
    use OCP\IUserManager;
    use Symfony\Component\Console\Input\InputArgument;
    use Symfony\Component\Console\Input\InputInterface;
    use Symfony\Component\Console\Input\InputOption;
    use Symfony\Component\Console\Output\OutputInterface;

    class ImportUsersCommand extends Base {
        private const SYNC_THRESHOLD = 50;

        public function __construct(
            private readonly IUserManager  $userManager,
            private readonly IGroupManager $groupManager,
            private readonly AuditService  $auditService,
            private readonly IJobList      $jobList,
        ) {
            parent::__construct();
        }

        protected function configure(): void {
            $this->setName('learning:import-users')
                ->setDescription('Import NC users from a CSV file and optionally assign to a NC group')
                ->addArgument('csv_file', InputArgument::REQUIRED, 'Path to CSV file (columns: username, display_name, email)')
                ->addOption('group', 'g', InputOption::VALUE_OPTIONAL, 'NC group GID to add users to');
            parent::configure();
        }

        protected function execute(InputInterface $input, OutputInterface $output): int {
            $csvFile = (string)$input->getArgument('csv_file');
            $group   = $input->getOption('group');
            $adminUid = \OC::$server->getUserSession()->getUser()?->getUID() ?? 'occ';

            if (!is_file($csvFile)) {
                $output->writeln("<error>CSV file not found: $csvFile</error>");
                return 1;
            }

            $csvContent = file_get_contents($csvFile);
            if ($csvContent === false) {
                $output->writeln('<error>Cannot read CSV file</error>');
                return 1;
            }

            $rows = array_filter(
                str_getcsv($csvContent, "\n"),
                fn(string $l): bool => trim($l) !== ''
            );

            // Skip header row
            $header = array_shift($rows);
            $dataRows = array_values($rows);
            $count = count($dataRows);

            if ($count > self::SYNC_THRESHOLD) {
                // Async path: dispatch QueuedJob — passwords generated inside the job
                $this->jobList->add(ImportUsersJob::class, [
                    'csv_data'  => implode("\n", array_merge([$header], $dataRows)),
                    'group'     => $group,
                    'admin_uid' => $adminUid,
                ]);
                $output->writeln("<info>Dispatched bulk import of $count users as background job.</info>");
                $output->writeln('<comment>Users will be created with random passwords. Use NC admin panel to send password-reset emails.</comment>');
                return 0;
            }

            // Sync path: create users immediately, print passwords once
            foreach ($dataRows as $line) {
                $cols = str_getcsv($line);
                $uid  = trim($cols[0] ?? '');
                $name = trim($cols[1] ?? $uid);
                if ($uid === '') {
                    continue;
                }

                // Generate random password — printed once, then discarded
                $password = bin2hex(random_bytes(12)); // 24 hex chars
                $user = $this->userManager->createUser($uid, $password);
                if ($user === false || $user === null) {
                    $output->writeln("<error>Failed to create user: $uid</error>");
                    continue;
                }

                if ($name !== '') {
                    $user->setDisplayName($name);
                }

                if ($group !== null) {
                    $grp = $this->groupManager->get($group);
                    $grp?->addUser($user);
                }

                $this->auditService->logEvent('user.imported', $adminUid, ['username' => $uid]);
                $output->writeln("<info>Created user: $uid  Password: $password</info>");
                unset($password); // explicitly clear from scope
            }

            return 0;
        }
    }
    ```

    Create app/lib/Command/ImportUsersJob.php:
    ```php
    <?php
    declare(strict_types=1);
    namespace OCA\Learning\Command;

    use OCA\Learning\Service\AuditService;
    use OCP\AppFramework\Utility\ITimeFactory;
    use OCP\BackgroundJob\QueuedJob;
    use OCP\IGroupManager;
    use OCP\IUserManager;

    /**
     * Background job for bulk NC user import (dispatched when CSV > 50 rows).
     *
     * SECURITY: Passwords are generated INSIDE this job, not in the job args.
     * Job args: ['csv_data' => string, 'group' => string|null, 'admin_uid' => string]
     * No 'password' key in args — job args persist in oc_jobs DB table.
     */
    class ImportUsersJob extends QueuedJob {
        public function __construct(
            private readonly IUserManager  $userManager,
            private readonly IGroupManager $groupManager,
            private readonly AuditService  $auditService,
            ITimeFactory $timeFactory,
        ) {
            parent::__construct($timeFactory);
        }

        /**
         * @param array{csv_data: string, group: string|null, admin_uid: string} $argument
         */
        protected function run(mixed $argument): void {
            $csvData  = (string)($argument['csv_data'] ?? '');
            $group    = isset($argument['group']) && $argument['group'] !== '' ? (string)$argument['group'] : null;
            $adminUid = (string)($argument['admin_uid'] ?? 'occ');

            $lines = array_filter(
                explode("\n", $csvData),
                fn(string $l): bool => trim($l) !== ''
            );
            array_shift($lines); // skip header

            foreach ($lines as $line) {
                $cols = str_getcsv($line);
                $uid  = trim($cols[0] ?? '');
                $name = trim($cols[1] ?? $uid);
                if ($uid === '') {
                    continue;
                }

                // Password generated here — never from job args (security: args persist in DB)
                $password = bin2hex(random_bytes(12));
                $user = $this->userManager->createUser($uid, $password);
                unset($password); // clear immediately after use

                if ($user === false || $user === null) {
                    continue;
                }
                if ($name !== '') {
                    $user->setDisplayName($name);
                }
                if ($group !== null) {
                    $grp = $this->groupManager->get($group);
                    $grp?->addUser($user);
                }

                $this->auditService->logEvent('user.imported', $adminUid, ['username' => $uid]);
            }
        }
    }
    ```
